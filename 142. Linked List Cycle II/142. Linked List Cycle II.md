Step1
LinkedList Cycleと同じような問題
setで重複なくノードを見ていって、同じ物が来たらそれを返せばいい
計算量O(n),空間計算量はO(n)となる

```python
# Definition for singly-linked list.
# class ListNode:
#     def __init__(self, x):
#         self.val = x
#         self.next = None


class Solution:
    """
    find the node where the cycle begins. If there is no cycle, return None.
    """

    def detectCycle(self, head: Optional[ListNode]) -> Optional[ListNode]:
        """
        store visited nodes , if node is already visited, return it.
        """
        visited = set()
        node = head

        while node is not None:
            if node in visited:
                return node
            visited.add(node)
            node = node.next

        return None

```

class Solution:
    """
    find the node where the cycle begins. If there is no cycle, return None.
    """

    def detectCycle(self, head: Optional[ListNode]) -> Optional[ListNode]:
        """
        find the node where the cycle begins. If there is no cycle, return None.
        """
        node = head
        collisionNode = self.collisionPoint(head)
        if collisionNode is None:
            return None

        while node is not collisionNode:
            node = node.next
            collisionNode = collisionNode.next

        return node

    def collisionPoint(self, head: Optional[ListNode]) -> Optional[ListNode]:
        """
        find the cycle. If there is no cycle, return None.
        """
        slow = head
        fast = head

        while fast is not None and fast.next is not None:
            slow = slow.next
            fast = fast.next.next
            if fast is slow:
                return slow

        return None
