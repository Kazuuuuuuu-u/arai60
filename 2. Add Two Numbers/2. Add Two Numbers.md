2. Add Two Numbers

最初に思ったこと
二つ考えた
一つは完全に一度数字に直す
計算してまたリストにする
２つ目は一つずつ計算して繰り上げを次に足していく
時間計算量はO(N)
空間はO（N）
一個目でやる
```python
# Definition for singly-linked list.
# class ListNode:
#     def __init__(self, val=0, next=None):
#         self.val = val
#         self.next = next
class Solution:
    def addTwoNumbers(
        self, l1: Optional[ListNode], l2: Optional[ListNode]
    ) -> Optional[ListNode]:
        if not l1 or not l2:
            return None
        num1 = self.to_number(l1)
        num2 = self.to_number(l2)
        sum_num = num1 + num2
        return self.to_reverse_ListNode_for_number(sum_num)

    def to_number(self, head: Optional[ListNode]) -> int:
        num = 0
        place = 1
        while head is not None:
            num += head.val * place
            place *= 10
            head = head.next
        return num

    def to_reverse_ListNode_for_number(self, num: int) -> Optional[ListNode]:
        if num == 0:
            return ListNode(0,None)

        node = None
        head = None
        while num > 0:
            digit = num % 10
            new_node = ListNode(digit,None)
            if node is None:
                node = new_node
                head = node
            else:
                node.next = new_node
                node = node.next
            num //= 10
        return head

```

３つぐらい考えた
一つは桁数が小さいノードに合わせてループを回す。桁数の小さい方を計算終わらせて後に桁上りの計算だけおこなう
２つ目は桁数が大きいノードに合わせる方法　ただ、ループ内にif文が増える
３つ目は桁数を合わせる方法 0で桁数が大きいのとおなじ桁数まで詰める。
３つ目を採用した
理由は計算のときに考えることが少なくて済むことで読む方も読みやすそうだから

```python
# class ListNode:
#     def __init__(self, val: int = 0, next: Optional["ListNode"] = None):
#         self.val = val
#         self.next = next
class Solution:
    def addTwoNumbers(
        self, l1: Optional[ListNode], l2: Optional[ListNode]
    ) -> Optional[ListNode]:
        if l1 is None or l2 is None:
            return None

        len1 = self.get_length(l1)
        len2 = self.get_length(l2)
        
        node = ListNode(-1, None)
        dummy = node
        if len1 < len2:
            l1 = self.padding(l1, len2 - len1, 0)
        elif len2 < len1:
            l2 = self.padding(l2, len1 - len2, 0)

        carry=0
        while l1 is not None and l2 is not None:
            sum_num=l1.val + l2.val + carry
            carry=sum_num // 10
            new_node=ListNode(sum_num % 10,None)
            node.next=new_node
            node=new_node
            l1=l1.next
            l2=l2.next

        if carry > 0:
            new_node=ListNode(carry)
            node.next=new_node
            node=new_node
            node.next=None

        return dummy.next

    def padding(
        self, head: Optional[ListNode], padding_count: int, padding_value: int
    ) -> Optional[ListNode]:
        node=head
        while node.next is not None:
            node=node.next

        for _ in range(padding_count):
            new_node=ListNode(padding_value,None)
            node.next=new_node
            node=new_node
        return head

    def get_length(self, head: Optional[ListNode]) -> int:
        length = 0
        while head:
            length += 1
            head = head.next
        return length
```
