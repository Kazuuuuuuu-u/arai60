20. Valid Parentheses

Step1
この問題を選んだときに

スタックに左側の括弧が
来たときはスタックに入れて、右側の括弧がきたらとりだしてそれに対応する括弧かどうかを調べる
時間計算量はO(N)

```python
class Solution:
    def isValid(self, s: str) -> bool:
        left_paren = '('
        right_paren = ')'
        left_brace = '{'
        right_brace = '}'
        left_bracket = '['
        right_bracket = ']'

        stack=[]
        for char in s:
            if char in (left_paren, left_brace, left_bracket):
                stack.append(char)
            else:
                if not stack:
                    return False
                top = stack.pop()
                if char == right_paren and top != left_paren:
                    return False
                if char == right_brace and top != left_brace:
                    return False
                if char == right_bracket and top != left_bracket:
                    return False
        
        return len(stack) == 0
```

```python
class Solution:
    def isValid(self, s: str) -> bool:
        open_to_close = {"(": ")", "{": "}", "[": "]"}
        open_brackets = []

        for char in s:
            if char in open_to_close.keys():
                open_brackets.append(char)
            elif char in open_to_close.values():
                open_bracket = open_brackets.pop()
                if not open_brackets or open_to_close[open_bracket] != char:
                    return False

        return len(open_brackets) == 0
```
