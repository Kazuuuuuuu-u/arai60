深さ優先だと考え方的には訪れたところに印をつけてそこに隣接するものをさらに訪れていく
もし訪れた場所だったらおわり
なので訪れた場所でDFSを呼び出して、他の人達に隣接島を探して印をつけてもらう。それが終わったらそこが一つの島なはず　なので最初のDFS呼び出しの回数を数えればいい

時間　２次元配列の大きさ分
空間　２次元配列の大きさ分
```python
class Solution:
    LAND = "1"
    WATER = "0"

    def numIslands(self, grid: List[List[str]]) -> int:

        row = len(grid)
        column = len(grid[0])
        visited = [[False] * column for _ in range(row)]
        island_count = 0

        def _explore_island(x, y):
            if not (0 <= x < row and 0 <= y < column):
                return
            if grid[x][y] == self.WATER:
                return
            if visited[x][y]:
                return

            visited[x][y] = True

            _explore_island(x - 1, y)
            _explore_island(x + 1, y)
            _explore_island(x, y - 1)
            _explore_island(x, y + 1)

        for i in range(row):
            for j in range(column):
                if grid[i][j] == self.LAND and not visited[i][j]:
                    _explore_island(i, j)
                    island_count += 1
        return island_count
```

```python
from collections import deque


class Solution:
    LAND = "1"
    WATER = "0"

    def numIslands(self, grid: List[List[str]]) -> int:
        row = len(grid)
        column = len(grid[0])
        visited = [[False] * column for _ in range(row)]
        num_islands = 0

        def _is_land(x: int, y: int) -> bool:
            if  0 <= x < row and 0 <= y < column:
                return grid[x][y] == self.LAND

        def _mark_connecting_lands_visited(x, y):
            to_be_traversed = deque()
            to_be_traversed.append((x, y))

            visited[x][y] = True

            while to_be_traversed:
                x, y = to_be_traversed.popleft()
                for dx, dy in ((-1, 0), (1, 0), (0, -1), (0, 1)):
                    neighbor_x = x + dx
                    neighbor_y = y + dy
                    if not _is_land(neighbor_x, neighbor_y):
                        continue
                    if visited[neighbor_x][neighbor_y]:
                        continue
                    to_be_traversed.append((neighbor_x, neighbor_y))
                    visited[neighbor_x][neighbor_y] = True

        for x in range(row):
            for y in range(column):
                if grid[x][y] == self.WATER:
                    continue

                assert grid[x][y] == self.LAND
                if visited[x][y]:
                    continue

                num_islands += 1
                _mark_connecting_lands_visited(x, y)

        return num_islands
```

```python
class Solution:
    LAND = "1"
    WATER = "0"

    def numIslands(self, grid: List[List[str]]) -> int:
        if not grid:
            return 0
        row = len(grid)
        column = len(grid[0])
        visited = [[False] * column for _ in range(row)]
        islands = []

        def _explore_island(i: int, j: int, island: list = None):
            if island is None:
                island = [[0] * column for _ in range(row)]
            if not (0 <= i < row and 0 <= j < column):
                return
            if grid[i][j] == self.WATER:
                return
            if visited[i][j]:
                return
            visited[i][j] = True
            island[i][j] = 1

            _explore_island(i + 1, j, island)
            _explore_island(i - 1, j, island)
            _explore_island(i, j + 1, island)
            _explore_island(i, j - 1, island)

            return island

        for i in range(row):
            for j in range(column):
                if not visited[i][j] and grid[i][j] == self.LAND:
                    islands.append(_explore_island(i, j))
        return len(islands)
```
