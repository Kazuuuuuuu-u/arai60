```python
class Solution:
    def topKFrequent(self, nums: List[int], k: int) -> List[int]:

        num_and_frequency = {}
        for num in nums:
            if num in num_and_frequency:
                num_and_frequency[num] += 1
            else:
                num_and_frequency[num] = 1
        num_and_frequency = sorted(
            num_and_frequency.items(), key=lambda x: x[1], reverse=True
        )

        return [num for num, _ in num_and_frequency[:k]]


```

ヒープとDefault　Dict
```python
from collections import defaultdict
from heapq import heappush, heappop


class Solution:
    def topKFrequent(self, nums: List[int], k: int) -> List[int]:
        if not 1 <= k <= len(nums):
            raise ValueError("k is invalid range")
        num_and_frequency = defaultdict(int)

        for num in nums:
            num_and_frequency[num] += 1

        topk_frequency_num = []

        for num, frequency in num_and_frequency.items():
            heappush(topk_frequency_num, (frequency, num))

        while len(topk_frequency_num) > k:
            heappop(topk_frequency_num)

        return [num for _, num in topk_frequency_num]

```

```python
from collections import defaultdict


class Solution:
    def topKFrequent(self, nums: List[int], k: int) -> List[int]:

        max_count=-1
        num_and_frequency=defaultdict(int)

        for num in nums:
            num_and_frequency[num]+=1
            max_count=max(max_count,num_and_frequency[num])

        sorted_for_frequency=[[] for _ in range(len(nums)+1)]
        for num,frequency in num_and_frequency.items():
            sorted_for_frequency[frequency].append(num)
        
        topk_frequency_nums=[]
        for i in range(max_count,-1,-1):
            if len(topk_frequency_nums)+len(sorted_for_frequency[i])<=k:
                topk_frequency_nums.extend(sorted_for_frequency[i])
                continue
            need_amount=k-len(topk_frequency_nums)
            topk_frequency_nums.extend(sorted_for_frequency[i][:need_amount])
            break

        return topk_frequency_nums
```
