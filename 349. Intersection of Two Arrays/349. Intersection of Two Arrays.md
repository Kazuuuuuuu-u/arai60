Step1
```python
class Solution:
    def intersection(self, nums1: List[int], nums2: List[int]) -> List[int]:

        nums1_without_duplication = set(nums1)
        nums2_without_duplication = set(nums2)

        intersection = []
        for num1 in nums1_without_duplication:
            if num1 in nums2_without_duplication:
                intersection.append(num1)

        return intersection
```

```python
class Solution:
    def intersection(self, nums1: List[int], nums2: List[int]) -> List[int]:

        if len(nums1) > len(nums2):
            smaller_nums_set = set(nums2)
            bigger_nums_list = nums1
        else:
            smaller_nums_set = set(nums1)
            bigger_nums_list = nums2

        intersections = []

        for num in bigger_nums_list:
            if num in smaller_nums_set:
                intersections.append(num)
                smaller_nums_set.remove(num)
        return intersections

```
