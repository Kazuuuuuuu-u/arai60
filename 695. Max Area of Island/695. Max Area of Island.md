```python
class Solution:
    WATER = 0
    LAND = 1

    def maxAreaOfIsland(self, grid: List[List[int]]) -> int:
        if not (grid or grid[0]):
            return 0
        grid_row = len(grid)
        grid_column = len(grid[0])
        visited = [[False] * grid_column for _ in range(grid_row)]

        def explore_island(row, column, island_nums=0):
            if not (0 <= row < grid_row and 0 <= column < grid_column):
                return island_nums
            if grid[row][column] == self.WATER:
                return island_nums
            if visited[row][column]:
                return island_nums

            visited[row][column] = True
            island_nums += 1

            island_nums = explore_island(row + 1, column, island_nums)
            island_nums = explore_island(row - 1, column, island_nums)
            island_nums = explore_island(row, column + 1, island_nums)
            island_nums = explore_island(row, column - 1, island_nums)
            return island_nums

        max_island_nums = 0
	for row in range(grid_row):
            for column in range(grid_column):
                if grid[row][column] == self.LAND:
                    max_island_nums = max(explore_island(row, column), max_island_nums)

        return max_island_nums

```
