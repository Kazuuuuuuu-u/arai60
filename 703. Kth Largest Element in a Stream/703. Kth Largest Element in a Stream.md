step1_1 : n_largest_scoresを使ったパターン
import heapq

```python
class KthLargest:

    def __init__(self, k: int, nums: List[int]):
        self.target = k
        self.n_largest_scores = heapq.nlargest(k, nums)
        heapq.heapify(self.n_largest_scores)

    def add(self, val: int) -> int:
        if len(self.n_largest_scores) < self.target:
            heapq.heappush(self.n_largest_scores, val)
        elif self.n_largest_scores[0] < val:
            heapq.heapreplace(self.n_largest_scores, val)
        return self.n_largest_scores[0]


# Your KthLargest object will be instantiated and called as such:
# obj = KthLargest(k, nums)
# param_1 = obj.add(val)
```

step1_2: 改善パターン そもそもn_largestが中でヒープを使っているならその後にheapifyするのは二度手間かも
N個のヒープを先に作ってK個になるまでpopするかK個のヒープを作って残りをヒープに入れるかどうか試すかのどちらか
シンプルさ的には前者と判断

```python
import heapq


class KthLargest:

    def __init__(self, k: int, nums: List[int]):
        self.target = k
        heapq.heapify(nums)
        for _ in range(len(nums) - k):
            heapq.heappop(nums)
        self.k_largest_scores = nums

    def add(self, val: int) -> int:
        if len(self.k_largest_scores) < self.target:
            heapq.heappush(self.k_largest_scores, val)
        elif self.k_largest_scores[0] < val:
            heapq.heapreplace(self.k_largest_scores, val)
        return self.k_largest_scores[0]


# Your KthLargest object will be instantiated and called as such:
# obj = KthLargest(k, nums)
# param_1 = obj.add(val)

```

```python
import heapq


class KthLargest:

    def __init__(self, k: int, nums: List[int]):
        self._k = k
        self._k_largest = []

        for num in nums:
            self.add(num)

    def add(self, val: int) -> int:

        heapq.heappush(self._k_largest, val)

        while len(self._k_largest) > self._k:
            heapq.heappop(self._k_largest)
        return self._k_largest[0]

# obj = KthLargest(k, nums)
# param_1 = obj.add(val)
```
