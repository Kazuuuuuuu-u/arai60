82. Remove Duplicates from Sorted List IIのstep3
指摘をもらったところを変えたバージョン
```python
# # Definition for singly-linked list.
# class ListNode:
#     def __init__(self, val=0, next=None):
#         self.val = val
#         self.next = next


class Solution:
    def deleteDuplicates(self, head: Optional[ListNode]) -> Optional[ListNode]:
        if head is None or head.next is None:
            return head

        node = head
        dummy = ListNode(0, None)
        tail_without_duplication = dummy

        while node is not None:
            if self.is_duplication_to_next(node):
                node = self.get_next_distinct_node(node)
            else:
                next_node = node.next
                node.next = None
                tail_without_duplication.next = node
                tail_without_duplication = node
                node = next_node
        return dummy.next

    def is_duplication_to_next(self, node):
        if node is None or node.next is None:
            return False
        return node.val == node.next.val

    def get_next_distinct_node(self, node):
        while self.is_duplication_to_next(node):
            node = node.next
        return node.next
```
