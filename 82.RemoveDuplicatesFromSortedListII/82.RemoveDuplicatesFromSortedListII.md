```python
# # Definition for singly-linked list.
# class ListNode:
#     def __init__(self, val=0, next=None):
#         self.val = val
#         self.next = next


class Solution:
    def deleteDuplicates(self, head: Optional[ListNode]) -> Optional[ListNode]:
        if not head or not head.next:
            return head

        dummy = ListNode(0)
        dummy.next = head
        node = head
        node_tail_without_duplication_with_dummy = dummy
        
        

        while node is not None:
            if self.is_duplicate(node):
                node = self.skip_duplicates(node)
            else:
                node_tail_without_duplication_with_dummy.next = node
                node_tail_without_duplication_with_dummy = (
                    node_tail_without_duplication_with_dummy.next
                )
                node = node.next
        node_tail_without_duplication_with_dummy.next = None

        return dummy.next

    def is_duplicate(self, node):
        if node.next is None:
            return False
        return node.val == node.next.val

    def skip_duplicates(self, node):
        while self.is_duplicate(node):
            node = node.next
        return node.next

```
