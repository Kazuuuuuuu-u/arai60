def deleteDuplicates(self, head: Optional[ListNode]) -> Optional[ListNode]:
        
        if not head:
            return None
        previous_val = head.val
        remove_duplicated_node = head
        node = head.next


        while node is not None:
            current_val = node.val
            if previous_val != current_val:
                previous_val = current_val
                remove_duplicated_node.next = node
                remove_duplicated_node = remove_duplicated_node.next
            else:
                remove_duplicated_node.next = None
            node = node.next

        return head

        return head

```python
def deleteDuplicates(self, head: Optional[ListNode]) -> Optional[ListNode]:
        
        if not head:
            return None

        node = head

        while node is not None and node.next is not None:
            if node.val == node.next.val:
                node.next = node.next.next
            else:
                node = node.next

        return head

```

def deleteDuplicates(self, head: Optional[ListNode]) -> Optional[ListNode]:
